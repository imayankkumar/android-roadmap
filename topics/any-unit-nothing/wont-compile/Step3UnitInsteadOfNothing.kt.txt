package com.imayankkumar.roadmap.anyunitnothing

fun stopOrderUnit(reason: String) {
    throw IllegalStateException(reason)
}

fun orderTotalAlmostRight(orderId: String): Int {
    val order = findOrder(orderId)
    if (order == null) stopOrderUnit("Order $orderId not found")
    return order.total
}
